# Pose assets + `pspr` (Mode 4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add optional `assets/poses.json` and a Mode-4-only `pspr` builtin that draws a pose as a list of 8x8 spritesheet cells with signed offsets and flips.

**Architecture:** Poses are ROM recipes, not pixel copies. Load JSON into `PoseRange`/`PosePart` tables, hang slice headers on `AssetImage`/`AssetMemory`, and implement `PSpr` by calling existing `Spr` per part so camera, clip, pal, and palt stay exact. Forced Mode 0 rejects `pspr`; AUTO falls back to Mode 4.

**Tech Stack:** Go, existing `assetsync` JSON loader, TinyGo-generated cart, Lua dialect builtins, `go test`.

**Spec:** `docs/superpowers/specs/2026-09-21-obj-pose-streaming-design.md` slice 1 only.

```go
func LoadPoses(projectDir string) (*PoseFile, error) {
	path := filepath.Join(projectDir, "assets", "poses.json")
	if _, err := os.Stat(path); errors.Is(err, os.ErrNotExist) {
		return &PoseFile{Version: 1}, nil
	} else if err != nil {
		return nil, fmt.Errorf("stat poses.json: %w", err)
	}
	var file PoseFile
	if err := readJSON(path, &file); err != nil {
		return nil, fmt.Errorf("read poses.json: %w", err)
	}
	if file.Version != 1 {
		return nil, fmt.Errorf("poses.json: version %d is unsupported (want 1)", file.Version)
	}
	if len(file.Poses) > MaxPoseCount {
		return nil, fmt.Errorf("poses.json: %d poses exceeds %d", len(file.Poses), MaxPoseCount)
	}
	names := make(map[string]int, len(file.Poses))
	total := 0
	for i, pose := range file.Poses {
		if pose.Name != "" {
			if prev, exists := names[pose.Name]; exists {
				return nil, fmt.Errorf("poses.json: duplicate pose name %q (indices %d and %d)", pose.Name, prev, i)
			}
			names[pose.Name] = i
		}
		if len(pose.Parts) > MaxPoseParts {
			return nil, fmt.Errorf("poses.json: pose %d has %d parts (maximum %d)", i, len(pose.Parts), MaxPoseParts)
		}
		total += len(pose.Parts)
		if total > MaxPosePartTotal {
			return nil, fmt.Errorf("poses.json: more than %d parts", MaxPosePartTotal)
		}
	}
	return &file, nil
}
```

```go
func (c *Context) PSpr(id, x, y int, flipX, flipY bool) {
	if c == nil || c.cart == nil || id < 0 || id >= len(c.cart.PoseIndex) {
		return
	}
	r := c.cart.PoseIndex[id]
	end := int(r.Start) + int(r.Count)
	if int(r.Start) > len(c.cart.PoseParts) || end > len(c.cart.PoseParts) {
		return
	}
	for _, part := range c.cart.PoseParts[r.Start:end] {
		ox, oy := int(part.X), int(part.Y)
		partFlipX := part.Flags&PoseFlipX != 0
		partFlipY := part.Flags&PoseFlipY != 0
		if flipX {
			ox = -int(part.X) - 8
			partFlipX = !partFlipX
		}
		if flipY {
			oy = -int(part.Y) - 8
			partFlipY = !partFlipY
		}
		c.Spr(int(part.Tile), x+ox, y+oy, 1, 1, partFlipX, partFlipY)
	}
}
```
